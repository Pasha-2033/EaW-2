	modifier = {
		set_temp_variable = {
			radiation_reduction = {
				value = radiation_value
				root = 3
				multiply = {
					value = 1
					add = modifier@radiation_cleanup
				}
				multiply = -1
			}
		}
		add = radiation_reduction
	}
}
