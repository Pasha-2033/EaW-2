scripted_map_modes = {
	building_slot_map_mode = {
		top = {
            type = state
            color = {
				FROM = {
					if = {
						limit = {
							check_variable = {
								common_all_slots > 0
							}
						}
						set_temp_variable = {
							slot_coefficient = {
								value = common_all_slots
								divide = 50
								root = 2
							}
						}
						set_temp_variable = {
							red = {
								value = 0.655
								multiply = slot_coefficient
							}
						}
						set_temp_variable = {
							green = {
								value = 0.96
								multiply = slot_coefficient
							}
						}
						set_temp_variable = {
							blue = {
								value = 0.6
								multiply = slot_coefficient
							}
						}
						set_temp_variable = {
							alpha = 1
						}
					}
					else = {
						set_temp_variable = { red = 0 }
						set_temp_variable = { green = 0.02 }
						set_temp_variable = { blue = 0.4 }
						set_temp_variable = { alpha = 0.7 }
					}
				}
            }
        }
		bottom = {
			type = none
		}
        far_text = none
        near_text = state
    }
}
