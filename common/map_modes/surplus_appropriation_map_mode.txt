scripted_map_modes = {
	surplus_appropriation_map_mode = {
		top = {
			type = none
		}
		bottom = {
            type = country
			color = {
				FROM = {
					if = {
						limit = {
							has_country_flag = no_sa_impact
						}
						set_temp_variable = { red = 0.1 }
						set_temp_variable = { green = 0.1 }
						set_temp_variable = { blue = 0.1 }
						set_temp_variable = { alpha = 1 }
					}
					else = {
						set_temp_variable = {
							temp = {
								value = surplus_appropriation_mod_values^0
								add = surplus_appropriation_mod_values^1
								add = surplus_appropriation_mod_values^0
								divide = 15 #because medium = 5, and we sum all of them
								subtract = 1 #<0 - below medium, >0 - above
							}
						}
						if = {
							limit = {
								check_variable = {
									temp < 0
								}
							}
							multiply_temp_variable = {
								temp = -1
							}
							set_temp_variable = { red = 1 }
							set_temp_variable = { green = 0 }
						}
						else = {
							set_temp_variable = { red = 0 }
							set_temp_variable = { green = 1 }
						}
						set_temp_variable = { blue = 0 }
						set_temp_variable = { alpha = temp }
					}
				}
			}
		}
		far_text = country
        near_text = none
	}
}
