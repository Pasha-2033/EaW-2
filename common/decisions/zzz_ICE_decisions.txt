ice_equipment_manpower_collecting = {
	open_tomb = {
		allowed = {
			original_tag = ICE
		}
		days_re_enable = open_tomb_cooldown
        cost = open_tomb_cost
		complete_effect = {
			add_equipment_to_stockpile = {
				type = windigo_equipment
				amount = open_tomb_eq_ammont
			}
		}
		ai_will_do = {
            base = 5000
        }
	}
	destroy_buildings = {
		allowed = {
			original_tag = ICE
		}
		state_target = yes
		target_root_trigger = {
			has_completed_focus = ice_focus_destroy_buildings_for_equipment
		}
		target_array = controlled_states
		on_map_mode = map_and_decisions_view
		days_re_enable = 14
		target_trigger = {
			FROM = {
				check_expr = {
					value = arms_factory_level
					add = industrial_complex_level
					add = building_level@agrarian_complex
					greater_than = 0
				}
			}
		}
		days_re_enable = destroy_buildings_cooldown
		cost = 10 #destroy_buildings_cost
		complete_effect = {
			set_temp_variable = {
				eq_num = {
					value = FROM.arms_factory_level
					multiply = 250
					add = {
						value = FROM.building_level@agrarian_complex
						multiply = 50
					}
					add = {
						value = FROM.industrial_complex_level
						multiply = 100
					}
				}
			}
			add_equipment_to_stockpile = {
				type = windigo_equipment
				amount = eq_num
			}
			FROM = {
				#"if" isn`t good, but for tooltip - we should leave them (remove text like "remove 0 building")
				if = {
					limit = {
						check_variable = {
							arms_factory_level > 0
						}
					}
					remove_building = {
						type = arms_factory
						level = arms_factory_level
					}
				}
				if = {
					limit = {
						check_variable = {
							building_level@agrarian_complex > 0
						}
					}
					remove_building = {
						type = agrarian_complex
						level = building_level@agrarian_complex
					}
				}
				if = {
					limit = {
						check_variable = {
							industrial_complex_level > 0
						}
					}
					remove_building = {
						type = industrial_complex
						level = industrial_complex_level
					}
				}
			}
		}
	}
	resurrect_fallen = {
		allowed = {
			original_tag = ICE
		}
		days_re_enable = resurrect_fallen_cooldown
        cost = resurrect_fallen_cost
		complete_effect = {
			set_temp_variable = {
				mp = {
					value = casualties_k
					subtract = resurrect_fallen_prev_casualties_k
					multiply = resurrect_fallen_efficiency
					multiply = 1000
				}
			}
			add_manpower = mp
			set_variable = {
				resurrect_fallen_prev_casualties_k = casualties_k
			}
		}
		ai_will_do = {
            base = 5000
        }
	}
}
