#rule_group_leader_industry
change_leader_rule_resource_industry = {
	type = change_leader_rules
	trigger = {
		faction_leader = {
			custom_override_tooltip = {
				tooltip = "has_resource_industry_superiority_tt"
				check_variable = {
					PREV.resource_industry_influence > resource_industry_influence
				}
			}
		}
	}
}

change_leader_rule_industry = {
	type = change_leader_rules
	trigger = {
		has_industry_to_become_leader = yes
	}
	ai_will_do = {
		base = 0
		modifier = {  
			add = 5 
			is_faction_leader = yes 
			OR = {
				industrial_complex > 100
				arms_factory > 100
			}
		}
	}
}

change_leader_rule_never = {
	type = change_leader_rules
	trigger = {
		always = no
	}
	available = {
		faction_manifest_fulfillment > 0.66
	}

	can_remove = { 
		faction_manifest_fulfillment < 0.5 
	}

	modifier = {
		research_sharing_per_country_bonus_factor = -0.25
		surrender_limit = -0.05
	}
	ai_will_do = {
		base = 0
		modifier = {  
			add = 5
			has_government = fascism
			is_faction_leader = yes 
		}
	}
}
