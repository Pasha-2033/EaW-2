#Low Scientist XP gain
sp_nuclear_generic_reward_scientist_xp_1 = {
	fire_only_once = no
	threshold = {
		min = 0
		max = 100
	}
	weight = {
		base = @base_reward_weight
		modifier = {
			factor = @reward_weight_factor_theme
			has_project_flag = sp_nuclear_generic_reward_scientist_xp_flag
		}
		modifier = {
			factor = 0
			var:scientist = {
				has_scientist_level = {
					level > @scientist_lvl_border
					specialization = specialization_nuclear
				}
			}
		}
	}
	option = {
		token = sp_nuclear_generic_reward_option_scientist_xp
		iteration_output = {
			scientist_effects = {
				if = {
					limit = {
						has_scientist_level = {
							level < @scientist_max_lvl
							specialization = specialization_nuclear
						}
					}
					add_scientist_xp = {
						experience = constant:sp_scientist_xp_gain.low
						specialization = specialization_nuclear
					}
				}
				FROM = {
					set_project_flag = sp_nuclear_generic_reward_scientist_xp_flag 
				}
			}
		}
	}
}
