@base_reward_weight = 2 #Base Reward weight value
@reward_weight_factor_theme = 0.25 # Applied to factor - Used to reduce chances of repeating the same reward theme within the same project (ie. Failure 1 & Failure 2)
@scientist_max_lvl = 5
@scientist_lvl_border = 4 #should be = scientist_max_lvl - 1
#Low Scientist XP gain
sp_industry_generic_reward_scientist_xp_1 = {
	fire_only_once = no
	threshold = {
		min = 0
		max = 100
	}
	weight = {
		base = @base_reward_weight
		modifier = {
			factor = @reward_weight_factor_theme
			has_project_flag = sp_industry_generic_reward_scientist_xp_flag
		}
		modifier = {
			factor = 0
			var:scientist = {
				has_scientist_level = {
					level > @scientist_lvl_border
					specialization = specialization_industry
				}
			}
		}
	}
	option = {
		token = sp_industry_generic_reward_option_scientist_xp
		iteration_output = {
			scientist_effects = {
				if = {
					limit = {
						has_scientist_level = {
							level < @scientist_max_lvl
							specialization = specialization_industry
						}
					}
					add_scientist_xp = {
						experience = constant:sp_scientist_xp_gain.low
						specialization = specialization_industry
					}
				}
				FROM = { 
					set_project_flag = sp_industry_generic_reward_scientist_xp_flag 
				}
			}
		}
	}
}

#Political interference
sp_industry_major_political_interference = {
	fire_only_once = yes
	threshold = { 
		min = 15
		max = 85
	}
	weight = {
		base = 0.25
		modifier = {
			factor = 4
			FROM = {
				OR = { 
					has_government = communism
					has_government = socialism
					has_government = radicalism
				}
			}
		}
		modifier = {
			#prevent loosing country leader
			var:scientist = {
				can_be_country_leader = yes
			}
			factor = 0
		}
	}
	option = {
		token = sp_industry_major_political_interference_option_keep
		iteration_output = {
			country_effects = {
				add_political_power = -50
				FROM = { 
					add_project_progress_ratio = constant:sp_progress.loss.medium
				}
			}
			scientist_effects = {
				if = {
					limit = {
						has_scientist_level = {
							level < @scientist_max_lvl
							specialization = specialization_industry
						}
					}
