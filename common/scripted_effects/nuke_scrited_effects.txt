damage_units_by_strike = {
	set_temp_variable = {
		damage_to_units_ratio = {
			value = 1
			subtract = var:target_state:modifier@nuke_defence_factor
			multiply = damage_multiplier?1
		}
	}
	set_temp_variable = {
		org_damage_to_units = {
			value = 0.5
			add = var:actor_country:modifier@nuke_charge_power_factor
			multiply = damage_to_units_ratio
			multiply = damage_multiplier?1
		}
	}
	set_temp_variable = {
		str_damage_to_units = {
			value = org_damage_to_units
			divide = 4
		}
	}
	#damage to enemy
	damage_units = {
		state = var:target_state
		org_damage = org_damage_to_units
		str_damage = str_damage_to_units
		limit = {
			has_war_with = var:actor_country
		}
		ratio = yes
		army = yes
		navy = yes
	}
	divide_temp_variable = {
		str_damage_to_units = 2
	}
	#damage to neutral/allies/yourself
	damage_units = {
		state = var:target_state
		org_damage = org_damage_to_units
		str_damage = str_damage_to_units
		limit = {
			NOT = {
				has_war_with = var:actor_country
			}
		}
		ratio = yes
		army = yes
		navy = yes
	}
}
nuke_population = {
	set_temp_variable_to_random = {
		var = state_nuked_population_k
		min = -0.25
		max = -0.05
	}
	set_temp_variable = {
		state_nuked_population_k = {
			value = state_nuked_population_k
			multiply = var:target_state:state_population_k
			multiply = {
				value = 1
				subtract = var:target_state:modifier@nuke_defence_factor
				add = var:actor_country:modifier@nuke_charge_power_factor
			}
			multiply = damage_multiplier?1
		}
	}
	var:target_state = {
		#PDX isn`t currently supports mul & div for 64x (it will overflow)
		for_loop_effect = {
			end = 1000
			add_manpower = state_nuked_population_k
		}
	}
}
#anti_nuke_bunker destructions (PREV should be country that dropped nuke)
nuke_anb = {
	set_temp_variable = {
		anb_level = {
			value = var:target_state:building_level@anti_nuke_bunker
			subtract = {
				value = random
				multiply = {
					value = var:actor_country:modifier@nuke_charge_power_factor
					add = 1
				}
				multiply = damage_multiplier?1
				max = 1
			}
			round = yes
		}
	}
	var:target_state = {
		remove_building = {
			type = anti_nuke_bunker
			level = building_level@anti_nuke_bunker
		}
		set_building_level = {
			type = anti_nuke_bunker
			level = anb_level
			instant_build = yes
		}
	}
}
#construction sites are not really buildings - they ignore damage
restore_construction_site = {
	var:target_state = {
		set_temp_variable = {
			building_level = building_level@construction_site
		}
		remove_building = {
			type = construction_site
			level = building_level
		}
		set_building_level = {
			type = construction_site
			level = building_level
			instant_build = yes
		}
	}
}
