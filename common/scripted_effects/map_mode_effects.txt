calculate_max_population = {
	every_state = {
		if = {
			limit = {
				check_variable = {
					state_population_k > global.max_population
				}
			}
			set_variable = {
				global.max_population = state_population_k
			}
		}
	}
}
calculate_all_states_building_slots = {
	every_state = {
		calculate_state_building_slots = yes
	}
}
calculate_state_building_slots = {
	set_variable = {
		common_builded_slots = {
			value = building_level@aluminium_plant
			add = building_level@steel_plant
			add = building_level@tungsten_plant
			add = building_level@chromium_plant
			add = building_level@ore_preparation
			add = building_level@recruiting_station
			add = building_level@labor_camp
			add = building_level@arms_factory
			add = building_level@industrial_complex
			add = building_level@agrarian_complex
			add = building_level@stronghold_network
			add = building_level@dockyard
			add = building_level@synthetic_refinery
			add = building_level@fuel_silo
			add = building_level@rocket_site
			add = building_level@mega_gun_emplacement
			add = building_level@nuclear_reactor
			add = building_level@nuclear_reactor_heavy_water
			add = building_level@nuclear_reactor_exploratory
			add = building_level@thermonuclear_reactor
			add = building_level@commercial_reactor
			add = building_level@social_infrastructure
		}
	}
	for_loop_effect = {
		start = 1
		end = 50
		value = i
		if = {
			limit = {
				meta_trigger = {
					text = {
						free_building_slots = {
							building = arms_factory
							include_locked = no
							size = [INDEX]
						}
					}
					INDEX = i
				}
			}
			set_variable = {
				common_empty_slots = i
			}
			set_temp_variable = {
				break = 1
			}
		}
	}
	set_variable = {
		common_all_slots = common_builded_slots
	}
	add_to_variable = {
		common_all_slots = common_empty_slots
	}
}
