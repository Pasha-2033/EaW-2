#THIS SCRIPTS WORKS ONLY IN STATE SCOPE!!!
add_state_manpower = {
	set_temp_variable = {
		state_pop_to_add = {
			value = modifier@social_infrastructure_num
			multiply = {
				value = 1
				add = owner:modifier@state_population_factor_by_social_infrastructure
			}
			add = modifier@state_population_factor
			min = 0
			multiply = state_population
		}
	}
	add_manpower = state_pop_to_add
}
