#THIS SCRIPTS WORKS ONLY IN STATE SCOPE!!!
add_state_manpower = {
	set_temp_variable = {
		state_pop_to_add = {
			value = modifier@social_infrastructure_num
			multiply = owner:modifier@state_population_factor_by_social_infrastructure
			clamp = { min = 0 max = 1000 } #max must be in clamp (crash otherwise, so 1000 is "reasonable" maximum)
			add = owner:modifier@state_population_factor
			multiply = state_population
		}
	}
	add_manpower = state_pop_to_add
}
