calc_coal_impact = {
	#energy 
	set_temp_variable = {
		energy_by_coal_factor = 1
	}
	add_to_temp_variable = {
		energy_by_coal_factor = modifier@energy_gain_factor
	}
	#restore actual coal value
	set_variable = {
		industry_efficiency_by_coal = resource@coal
	}
	add_to_variable = {
		industry_efficiency_by_coal = total_coal_by_all
	}
	#total coal setup
	set_variable = {
		total_coal_by^0 = 0
	}
	set_variable = {
		total_coal_by^1 = 0
	}
	set_variable = {
		total_coal_by^2 = 0
	}
	set_variable = {
		total_coal_by^3 = 0
	}
	set_variable = {
		total_coal_by_synthetic = 0
	}
	#iterate states
	every_controlled_state = {
		#local energy consumption factor
		set_temp_variable = {
			local_energy_consumption = 1
		}
		add_to_temp_variable = {
			local_energy_consumption = modifier@local_factory_energy_consumption
		}
		#local energy consumption factor (by infrastructure)
		set_temp_variable = {
			local_energy_per_infra = modifier@local_factory_energy_consumption_per_infrastructure
		}
		multiply_temp_variable = {
			local_energy_per_infra = non_damaged_building_level@infrastructure
		}
		#summarise local_energy
		add_to_temp_variable = {
			local_energy_consumption = local_energy_per_infra
		}
		clamp_temp_variable = {
			var = local_energy_consumption
			min = 0
		}
		#aluminium
		set_variable = {
			state_aluminium_plant_impact = local_energy_consumption
		}
		multiply_variable = {
			state_aluminium_plant_impact = non_damaged_building_level@aluminium_plant
		}
		multiply_variable = {
			state_aluminium_plant_impact = PREV.modifier@aluminium_plant_coal_consumption
		}
		multiply_variable = {
			state_aluminium_plant_impact = PREV.plant_max_power^0
		}
		add_to_variable = {
			PREV.total_coal_by^0 = state_aluminium_plant_impact
		}
		set_variable = {
			#init total state consumption
			state_coal_consumption = state_aluminium_plant_impact
		}
		#steel
		set_variable = {
			state_steel_plant_impact = local_energy_consumption
		}
		multiply_variable = {
			state_steel_plant_impact = non_damaged_building_level@steel_plant
		}
		multiply_variable = {
			state_steel_plant_impact = PREV.modifier@steel_plant_coal_consumption
		}
		multiply_variable = {
			state_steel_plant_impact = PREV.plant_max_power^1
		}
		add_to_variable = {
			PREV.total_coal_by^1 = state_steel_plant_impact
		}
		add_to_variable = {
			state_coal_consumption = state_steel_plant_impact
		}
		#tungsten
		set_variable = {
			state_tungsten_plant_impact = local_energy_consumption
		}
		multiply_variable = {
			state_tungsten_plant_impact = non_damaged_building_level@tungsten_plant
		}
		multiply_variable = {
			state_tungsten_plant_impact = PREV.modifier@tungsten_plant_coal_consumption
		}
		multiply_variable = {
			state_tungsten_plant_impact = PREV.plant_max_power^2
		}
		add_to_variable = {
			PREV.total_coal_by^2 = state_tungsten_plant_impact
		}
		add_to_variable = {
			state_coal_consumption = state_tungsten_plant_impact
		}
		#chromium
		set_variable = {
			state_chromium_plant_impact = local_energy_consumption
		}
		multiply_variable = {
			state_chromium_plant_impact = non_damaged_building_level@chromium_plant
		}
		multiply_variable = {
			state_chromium_plant_impact = PREV.modifier@chromium_plant_coal_consumption
		}
		multiply_variable = {
			state_chromium_plant_impact = PREV.plant_max_power^3
		}
		add_to_variable = {
			PREV.total_coal_by^3 = state_chromium_plant_impact
		}
		add_to_variable = {
			state_coal_consumption = state_chromium_plant_impact
		}
		#refinery
		set_variable = {
			state_synthetic_refinery_impact = local_energy_consumption
		}
		multiply_variable = {
			state_synthetic_refinery_impact = non_damaged_building_level@synthetic_refinery
		}
		multiply_variable = {
			state_synthetic_refinery_impact = PREV.modifier@synthetic_refinery_coal_consumption
		}
		multiply_variable = {
			state_synthetic_refinery_impact = PREV.synthetic_refinery_max_power
		}
		add_to_variable = {
			PREV.total_coal_by_synthetic = state_synthetic_refinery_impact
		}
		add_to_variable = {
			state_coal_consumption = state_synthetic_refinery_impact
		}
	}
	#total consumtion
	set_variable = {
		total_coal_by_all = total_coal_by^0
	}
	add_to_variable = {
		total_coal_by_all = total_coal_by^1
	}
	add_to_variable = {
		total_coal_by_all = total_coal_by^2
	}
	add_to_variable = {
		total_coal_by_all = total_coal_by^3
	}
	add_to_variable = {
		total_coal_by_all = total_coal_by_synthetic
	}
	#calc industry efficiency (by coal)
	divide_variable = {
		industry_efficiency_by_coal = total_coal_by_all
	}
	set_temp_variable = {
		#clamp can`t accept modifier
		min_value = modifier@industry_min_efficiency_by_coal
	}
	clamp_variable = {
		var = industry_efficiency_by_coal
		min = min_value
		max = 1
	}
}
calc_resource_aluminium = {
	#restore resource value
	set_variable = {
		resource_plant_efficiency^0 = resource@bauxite
	}
	add_to_variable = {
		resource_plant_efficiency^0 = resource_consumption^0
	}
	#impact (important to call before mtth!)
	set_variable = {
		aluminium_plant_total_impact = modifier@aluminium_plant_num
	}
	multiply_variable = {
		aluminium_plant_total_impact = plant_max_power^0
	}
	#consumtion
	set_variable = {
		resource_consumption^0 = mtth:aluminium_plant_total_bauxite_consumption
	}
	if = {
		limit = {
			check_variable = {
				resource_consumption^0 > 0
			}
		}
		#effeciency
		divide_variable = {
			resource_plant_efficiency^0 = resource_consumption^0
		}
		add_to_variable = {
			resource_plant_efficiency^0 = modifier@aluminium_plant_penalty_reduction
		}
		clamp_variable = {
			var = resource_plant_efficiency^0
			min = 0
			max = 1
		}
		#outcome
		set_variable = {
			resource_outcome^0 = mtth:aluminium_plant_total_aluminium_outcome
		}
		multiply_variable = {
			resource_outcome^0 = resource_plant_efficiency^0
		}
		multiply_variable = {
			resource_outcome^0 = -1
		}
	}
	else = {
		set_variable = {
			resource_plant_efficiency^0 = 1
		}
		set_variable = {
			resource_outcome^0 = 0
		}
	}
}

calc_primary_reactor = {
	#restore resource value
	set_variable = {
		reactor_primary_efficiency = resource@nuclear_fuel_primary
	}
	add_to_variable = {
		reactor_primary_efficiency = primary_fuel_reactor_consumption
	}
	#impact (important to call before mtth!)
	set_variable = {
		reactor_common_total_impact = modifier@nuclear_reactor_num
	}
	#multiply_variable = {
	#	reactor_common_total_impact = reactor_common_max_power
	#}
	set_variable = {
		reactor_exploratory_total_impact = modifier@nuclear_reactor_exploratory_num
	}
	#multiply_variable = {
	#	reactor_exploratory_total_impact = reactor_exploratory_max_power
	#}
	#consumtion
	set_variable = {
		primary_fuel_reactor_consumption = mtth:reactor_common_total_consumtion
	}
	add_to_variable = {
		primary_fuel_reactor_consumption = mtth:reactor_exploratory_total_consumtion
	}
	if = {
		limit = {
			check_variable = {
				primary_fuel_reactor_consumption > 0
			}
		}
		#effeciency
		divide_variable = {
			reactor_primary_efficiency = primary_fuel_reactor_consumption
		}
		set_variable = {
			#for own penalty reduction
			reactor_exploratory_primary_efficiency = reactor_primary_efficiency
		}
		add_to_variable = {
			reactor_primary_efficiency = modifier@nuclear_reactor_penalty_reduction
		}
		add_to_variable = {
			reactor_exploratory_primary_efficiency = modifier@nuclear_reactor_exploratory_penalty_reduction
		}
		clamp_variable = {
			var = reactor_primary_efficiency
			min = 0
			max = 1
		}
		clamp_variable = {
			var = reactor_exploratory_primary_efficiency
			min = 0
			max = 1
		}
		#outcome
		set_variable = {
			#set because +(-x), cost subtracts
			nuclear_fuel_primary_reactor = primary_fuel_reactor_consumption
		}
		set_variable = {
			nuclear_fuel_secondary_reactor = mtth:nuclear_reactor_total_fuel_secondary_outcome
		}
		set_variable = {
			common_thermonuclear_fuel_reactor = mtth:nuclear_reactor_total_fuel_thermonuclear_outcome
		}
		set_variable = {
			thermonuclear_fuel_reactor = mtth:nuclear_reactor_exploratory_total_fuel_thermonuclear_outcome
		}
		set_variable = {
			nuclear_production_factor_outcome = mtth:nuclear_reactor_total_nuke_outcome
		}
		set_variable = {
			thermonuclear_production_factor_outcome = mtth:nuclear_reactor_total_thermonuke_outcome
		}
		multiply_variable = {
			common_thermonuclear_fuel_reactor = reactor_primary_efficiency
		}
		multiply_variable = {
			nuclear_fuel_secondary_reactor = reactor_primary_efficiency
		}
		multiply_variable = {
			thermonuclear_fuel_reactor = reactor_exploratory_primary_efficiency
		}
		multiply_variable = {
			nuclear_production_factor_outcome = reactor_primary_efficiency
		}
		multiply_variable = {
			thermonuclear_production_factor_outcome = reactor_primary_efficiency
		}
		add_to_variable = {
			thermonuclear_fuel_reactor = common_thermonuclear_fuel_reactor
		}
		multiply_variable = {
			nuclear_fuel_secondary_reactor = -1
		}
		multiply_variable = {
			thermonuclear_fuel_reactor = -1
		}
	}
	else = {
		#fuel
		set_variable = {
			nuclear_fuel_primary_reactor = 0
		}
		set_variable = {
			nuclear_fuel_secondary_reactor = 0
		}
		set_variable = {
			thermonuclear_fuel_reactor = 0
		}
		#bomb production
		set_variable = {
			nuclear_production_factor_outcome = 0
		}
		set_variable = {
			thermonuclear_production_factor_outcome = 0
		}
	}
}

calc_secondary_reactor = {
	#restore resource value
	set_variable = {
		reactor_heavy_water_efficiency = resource@nuclear_fuel_secondary
	}
	add_to_variable = {
		reactor_heavy_water_efficiency = secondary_fuel_reactor_consumption
	}
	#impact (important to call before mtth!)
	set_variable = {
		reactor_heavy_water_total_impact = modifier@nuclear_reactor_heavy_water_num
	}
	#multiply_variable = {
	#	reactor_heavy_water_total_impact = reactor_heavy_water_max_power
	#}
	#consumtion
	set_variable = {
		secondary_fuel_reactor_consumption = mtth:secondary_fuel_reactor_total_consumption
	}
	if = {
		limit = {
			check_variable = {
				secondary_fuel_reactor_consumption > 0
			}
		}
		#effeciency
		divide_variable = {
			reactor_heavy_water_efficiency = secondary_fuel_reactor_consumption
		}
		add_to_variable = {
			reactor_heavy_water_efficiency = modifier@nuclear_reactor_heavy_water_penalty_reduction
		}
		clamp_variable = {
			var = reactor_heavy_water_efficiency
			min = 0
			max = 1
		}
		#outcome
		add_to_variable = {
			#add because +(-x), cost subtracts
			nuclear_fuel_secondary_reactor = secondary_fuel_reactor_consumption
		}
		set_variable = {
			secondary_nuke = mtth:secondary_total_nuke
		}
		set_variable = {
			secondary_thermonuke = mtth:secondary_total_thermonuke
		}
		set_variable = {
			primary_fuel = mtth:total_primary_fuel
		}
		set_variable = {
			heavy_water_fuel_thermonuclear_outcome = mtth:heavy_water_fuel_thermonuclear_total_outcome
		}
		multiply_variable = {
			heavy_water_fuel_thermonuclear_outcome = reactor_heavy_water_efficiency
		}
		multiply_variable = {
			secondary_nuke = reactor_heavy_water_efficiency
		}
		multiply_variable = {
			secondary_thermonuke = reactor_heavy_water_efficiency
		}
		multiply_variable = {
			primary_fuel = reactor_heavy_water_efficiency
		}
		add_to_variable = {
			nuclear_production_factor_outcome = secondary_nuke
		}
		add_to_variable = {
			thermonuclear_production_factor_outcome = secondary_thermonuke
		}
		subtract_from_variable = {
			#sub because -(-x), cost subtracts
			nuclear_fuel_primary_reactor = primary_fuel
		}
		subtract_from_variable = {
			#sub because -(-x), cost subtracts
			thermonuclear_fuel_reactor = heavy_water_fuel_thermonuclear_outcome
		}
	}
	else = {
		set_variable = {
			reactor_heavy_water_efficiency = 1
		}
	}
}
